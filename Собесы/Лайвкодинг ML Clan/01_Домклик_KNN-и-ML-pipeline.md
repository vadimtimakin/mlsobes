```python
def knn_predict(X_train, y_train, X_test, k=3):
    predictions = []
    for test_point in X_test:
        distances = [sum((x - test_point) ** 2) for x in X_train]  # Евклидово расстояние
        nearest = np.argsort(distances)[:k]
        top_labels = [y_train[i] for i in nearest]
        predictions.append(max(set(top_labels), key=top_labels.count))
    return predictions
```
