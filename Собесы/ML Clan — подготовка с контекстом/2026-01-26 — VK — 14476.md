> '''
> Дан 2-D лабиринт Maze размера M x N, в котором два типа точек:
>
> '.' - проходимые;
>
> '#' - непроходимые.
>
> Движение начинается в верхнем левом углу, а заканчивается в нижнем правом (стартовая и конечная точки всегда проходимые). 
> Допускаются движения вверх ("UP"), вниз ("DOWN"), влево ("LEFT") и вправо ("RIGHT").
>
> Задача
> Выдать список инструкций для кратчайшего пути.
> '''
>
> from typing import List
> maze = [
>   [".", ".", "#", "."],
>   ["#", ".", "#", "#"],
>   [".", ".", ".", "#"],
>   [".", "#", ".", "#"],
>   [".", ".", ".", "."],
>   [".", "#", ".", "."],
> ]
>
>
> from collections import queue
>
> def find_a_way_out(maze: List[List[str]]) -> List[str]:
>     m, n = len(maze), len(maze[0])
>     points = queue([(0, 0)])
>     parent = {(0, 0): None}
>     
>     while points:
>         x, y = points.popleft()
>         
>         new_points([(x+1, y), (x-1, y), (x, y+1), (x, y-1)])
>         for xn, yn in new_points:
>             if 0 <= xn < m and 0 <= yn < n and maze[xn][yn] == '.' and (xn, yn) not in parent:
>                 parent[(xn, yn)] = (x, y)
>                 points.append((xn, yn))
>             
>     if (m-1, n-1) not in parent:
>         return []
>     
>     instructions = []
>     directions = {(1, 0): "UP", (-1, 0): "DOWN", (0, 1): "LEFT", (0, -1): "RIGHT"}
>     
>     x, y = m-1, n-1
>     while parent[(x, y)] is not None:
>         xn, yn = parent[(x, y)]
>         instructions.append(directions[(xn-x, yn-y)])
>         x, y = xn, yn
>     return instructions[::-1]
>     
>         
> instructions = find_a_way_out(maze)  # ["RIGHT", "DOWN", "DOWN", "RIGHT", "DOWN", "DOWN", "DOWN", "RIGHT"]
