> def f(x):
>         return a * x**2 + b * x + c
>
>     def grad(x):
>         return 2*(a * x**2 + b * x + c)*2 * a * x + b  
>
>     x0 = -b / (2 * a) + 10.0
>     x1 = -b / (2 * a) - 10.0
>
>     for _ in range(1000000):
>         x0 = x0 - lr * grad(x0)
>         x1 = x1 - lr * grad(x1)
>
>     return (x0, x1)
