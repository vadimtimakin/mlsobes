> class LinearRegression:
>     def __init__(self, learning_rate: float = 0.01, epochs: int = 1000) -> None:
>         self.learning_rate = learning_rate
>         self.epochs = epochs
>
>         self.weights: list[float] = []
>         self.bias = 0.
>
>         self.is_fitted = False
>
>     def train(self, X: Sequence[Sequence[float]], y: Sequence[float]) -> None:
>         if (len(X) != len(y)):
>             raise ValueError("кол-во объектов должно совпадать с кол-вом целевых значений")
>
>         if not X:
>             raise ValueError("2")
>
>         #размерность X
>
>         n_samples = len(X)
>         n_features = len(X[0])
>
>         self.weights = [0.] * n_features
>         self.bias = 0.
>
>         for _ in range(self.epochs):
>             predictions = [
>                 sum(weight * feature for weight, feature in zip(self.weights, row)) + self.bias
>                 for row in X
>             ]
>
>             errors = [
>                 prediction - target for prediction, target in zip(predictions, y+)
>             ]
>
>             weight_gradient = [
>                 2. / n_samples * sum(error * X[row_index][feature_index] for row_index, error in enumerate(errors)) 
>                 for feature_index in range(n_features)
>             ]
>
>             bias_gradient = 2. / n_samples * sum(errors)
>
>             for feature_index in range(n_features):
>                 self.weights[feature_index] -= (self.learning_rate * weight_gradient[feature_index])
>
>             self.bias -= self.learning_rate * bias_gradient
>
>         self.is_fitted = True
>             
>     def predict(self, X: Sequence[Sequence[float]]) -> list[float]:
>         if not self.is_fitted:
>             raise ValueError("3")
>
>         if not X:
>             return []
>
>         return [
>             sum(weight * feature for weight, feature in zip(self.weights, row) + self.bias 
>             for row in X
>             )
>         ]

> # На вход приходит массив строк только с английскими буквами input: list[str], нужно сгруппировать
> # все анаграммы (любые слова, составленные с помощью перестановок исходных букв) в группы и вывести
> # список этих групп в произвольном порядке output: list[list[str]], пример:
> # Input: strs = ["eat", "tea", "tan", "ate", "nat", "bat"]
> from collection import defaultdict
>
> def group_anagrams(strs: list[str]) -> list[list[str]]:
>     groups: dict[tuple[int, ...], list[str]] = {}
>
>
>
>     for word in strs:
>         counts = [0] * 26
>
>         for char in word:
>             counts[ord(char) - ord("a")] += 1
>         
>         key = tuple(counts)
>
>         groups.setdefault(key, []).append(word)
>
>     return list(groups.value())
>
> O(n + (k + 26)) - space
> O(n * k) - time
