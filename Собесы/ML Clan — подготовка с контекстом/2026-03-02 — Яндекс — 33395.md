> from typing import List, Tuple

> def max_monotonic(nums: List[int]) -> Tuple[int, int]:
>     if not nums:
>         return (0, 0)
>     n = len(nums)
>     
>     # Инициализация
>     max_len = 1
>     max_start = 0
>     max_end = 0
>     
>     inc_start = 0  # старт неубывающего подотрезка
>     dec_start = 0  # старт невозрастающего подотрезка
>     
>     for i in range(1, n):
>         # Проверяем неубывающий
>         if nums[i] > nums[i - 1]:
>             # продолжаем подотрезок
>             pass
>         else:
>             inc_start = i
>         
>         # Проверяем невозрастающий
>         if nums[i] < nums[i - 1]:
>             pass
>         else:
>             dec_start = i
>         
>         # Вычисляем текущие длины
>         inc_len = i - inc_start + 1
>         dec_len = i - dec_start + 1
>         
>         # Обновляем максимум
>         if inc_len > max_len:
>             max_len = inc_len
>             max_start = inc_start
>             max_end = i
>         if dec_len > max_len:
>             max_len = dec_len
>             max_start = dec_start
>             max_end = i
>     
>     return (max_start, max_end)
>
> Время: O(n) — один проход по массиву
> Память: O(1) — константное дополнительное пространство
